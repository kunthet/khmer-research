  • ការចាត់ថ្នាក់ផលិតផលថ្មីទៅជា ៦ ប្រភេទ (ដូចជាផលិតផលថ្មីទាំងស្រុងសម្រាប់ពិភពលោក ឬការកាត់បន្ថយថ្លៃដើម) ជួយក្រុមហ៊ុនក្នុងការជ្រើសរើសយុទ្ធសាស្ត្រដែលសមស្របបំផុត។
  • ការជ្រើសរើសយុទ្ធសាស្ត្រអាជីវកម្ម (ដូចជាការអភិវឌ្ឍន៍ផលិតផល ឬការធ្វើពិពិធកម្ម) ត្រូវតែផ្អែកលើធនធានផ្ទៃក្នុងរបស់ក្រុមហ៊ុន និងកត្តាប្រកួតប្រជែងខាងក្រៅដើម្បីធានាបាននូវការរស់រានមានជីវិតរបស់អាជីវកម្ម។

២. ការវិភាគលើប្រសិទ្ធភាព និងការចំណាយ (Performance & Constraints)

វិធីសាស្ត្រ (Method) គុណសម្បត្តិ (Pros) គុណវិបត្តិ (Cons) លទ្ធផលគន្លឹះ (Key Result)
Radical Innovation (New-to-the-world products)
នវានុវត្តន៍បែបស៊ីជម្រៅ (ការបង្កើតផលិតផលថ្មីទាំងស្រុង)		 បង្កើតទីផ្សារថ្មីស្រឡាង និងផ្តល់នូវអត្ថប្រយោជន៍ប្រកួតប្រជែងខ្ពស់បំផុត។ មានហានិភ័យខ្ពស់ ទាមទារចំណេះដឹងផ្នែកបច្ចេកវិទ្យាទំនើប និងការវិនិយោគច្រើន។ ទាមទារការគ្រប់គ្រងបែបប្រមូលផ្តុំ (Centralization) ដើម្បីឱ្យការអនុវត្តមានប្រសិទ្ធភាព។
Incremental Innovation (Product Improvement)
នវានុវត្តន៍បន្តិចម្តងៗ (ការកែលម្អផលិតផលដែលមានស្រាប់)		 មានហានិភ័យទាប ដោយសារក្រុមការងារមានភាពស៊ាំ និងប្រើប្រាស់បច្ចេកវិទ្យាដែលមានស្រាប់។ អត្ថប្រយោជន៍ប្រកួតប្រជែងមិនសូវមាននិរន្តរភាព ដោយសារគូប្រជែងងាយស្រួលក្នុងការធ្វើតាម។ ការគ្រប់គ្រងបែបវិមជ្ឈការ (Decentralization) ផ្តល់លទ្ធផលល្អជាង ព្រោះវាជំរុញឱ្យបុគ្គលិកបញ្ចេញគំនិតច្នៃប្រឌិត។
Cost Reductions
ការកាត់បន្ថយថ្លៃដើមផលិតកម្ម		 ជួយបង្កើនប្រាក់ចំណេញ (Value Addition) ដល់ក្រុមហ៊ុនយ៉ាងច្រើន ទោះបីជាផលិតផលមានរូបរាងដូចដើមក៏ដោយ។ មិនបានផ្តល់នូវអត្ថប្រយោជន៍បន្ថែមថ្មីៗណាមួយដល់អតិថិជនដោយផ្ទាល់នោះទេ។ ជួយបង្កើនប្រសិទ្ធភាពនៃការគ្រប់គ្រងថ្លៃដើម (Efficiency of production cost management)។

ការចំណាយលើធនធាន (Resource Cost)៖ ឯកសារនេះជាប្រភេទអត្ថបទពិនិត្យទ្រឹស្តី ដែលមិនបានបញ្ជាក់ពីតម្រូវការផ្នែករឹង (Hardware) ឬផ្នែកទន់ (Software) ជាក់លាក់នោះទេ ប៉ុន្តែបានសង្កត់ធ្ងន់លើធនធានមនុស្ស រចនាសម្ព័ន្ធគ្រប់គ្រង និងថវិកា។

  • Expertise: ទាមទារក្រុមការងារផ្នែកស្រាវជ្រាវ និងអភិវឌ្ឍន៍ (R&D) និងអ្នកជំនាញផ្នែកទីផ្សារដែលមានចំណេះដឹងស៊ីជម្រៅ។
  • Dataset: ត្រូវការទិន្នន័យជាក់លាក់ស្តីពីអាកប្បកិរិយារបស់អតិថិជនគោលដៅ (Consumer Behavior) ដើម្បីកាត់បន្ថយហានិភ័យ។
  • Financial/Capital: ថវិកាសម្រាប់ការវិនិយោគលើម៉ាស៊ីន បច្ចេកវិទ្យាថ្មីៗ និងយុទ្ធនាការផ្សព្វផ្សាយទីផ្សារ។

៣. ការពិនិត្យសម្រាប់បរិបទកម្ពុជា/អាស៊ីអាគ្នេយ៍

ភាពលំអៀងនៃទិន្នន័យ (Data Bias)៖

ឯកសារនេះជាអត្ថបទស្រាវជ្រាវបែបទ្រឹស្តី (Literature Review) ដោយផ្អែកលើការលើកឡើងនូវករណីសិក្សារបស់ក្រុមហ៊ុនអន្តរជាតិធំៗ (ដូចជា 3M, Apple, Samsung, Honda) ដោយមិនមានការប្រមូលទិន្នន័យពីប្រជាសាស្ត្រណាមួយជាក់លាក់ឡើយ។ សម្រាប់ប្រទេសកម្ពុជា ការយកទ្រឹស្តីអន្តរជាតិទាំងនេះមកអនុវត្ត ត្រូវតែមានការកែសម្រួលឱ្យស្របទៅតាមបរិបទនៃកម្រិតចំណូល អំណាចទិញ និងចំណង់ចំណូលចិត្តរបស់ប្រជាជនក្នុងស្រុក។

លទ្ធភាពនៃការអនុវត្ត (Applicability)៖

ទ្រឹស្តី និងយុទ្ធសាស្ត្រនៅក្នុងអត្ថបទនេះ មានសារៈសំខាន់ និងអាចយកមកអនុវត្តបានយ៉ាងល្អសម្រាប់សហគ្រាស និងក្រុមហ៊ុននៅក្នុងប្រទេសកម្ពុជា។

  • វិស័យកសិឧស្សាហកម្មនៅខេត្តបាត់ដំបង (Agro-processing in Battambang): អាចប្រើប្រាស់យុទ្ធសាស្ត្រធ្វើពិពិធកម្ម (Concentric Diversification) ដើម្បីកែច្នៃកសិផលមូលដ្ឋានទៅជាផលិតផលថ្មីៗដែលមានតម្លៃបន្ថែមខ្ពស់។
  • សហគ្រាសធុនតូច និងមធ្យមនៅភ្នំពេញ (SMEs in Phnom Penh): អាចអនុវត្តយុទ្ធសាស្ត្រកាត់បន្ថយថ្លៃដើម (Cost Reductions) ដោយប្រើប្រាស់វត្ថុធាតុដើមកែច្នៃឡើងវិញ ដើម្បីប្រកួតប្រជែងតម្លៃជាមួយផលិតផលនាំចូល។
  • ធុរកិច្ចថ្មីផ្នែកបច្ចេកវិទ្យា (Tech Startups): អាចប្រើប្រាស់យុទ្ធសាស្ត្រនវានុវត្តន៍បែបស៊ីជម្រៅ (Radical Innovation) ដើម្បីបង្កើតដំណោះស្រាយបច្ចេកវិទ្យាថ្មីៗ (New-to-the-world) ដែលដោះស្រាយបញ្ហាជាក់លាក់នៅក្នុងសង្គមខ្មែរ។

ការជ្រើសរើស និងអនុវត្តយុទ្ធសាស្ត្រអភិវឌ្ឍន៍ផលិតផលឱ្យបានត្រឹមត្រូវ នឹងជួយឱ្យក្រុមហ៊ុននៅកម្ពុជាអាចកាត់បន្ថយហានិភ័យនៃការបរាជ័យ និងអាចពង្រីកចំណែកទីផ្សារប្រកបដោយនិរន្តរភាព។

៤. ផែនការសកម្មភាពសម្រាប់និស្សិត (Actionable Roadmap)

ដើម្បីអនុវត្តតាមការសិក្សានេះ និស្សិតគួរអនុវត្តតាមជំហានខាងក្រោម៖

  1. វិភាគស្ថានភាពទីផ្សារ និងកំណត់យុទ្ធសាស្ត្រ (Market Analysis & Strategy formulation): សិក្សាពីតម្រូវការទីផ្សារ និងគូប្រជែង ដោយប្រើប្រាស់ Ansoff Matrix ដើម្បីកំណត់ថាតើអាជីវកម្មគួរប្រើប្រាស់យុទ្ធសាស្ត្រអភិវឌ្ឍន៍ផលិតផល ជ្រៀតចូលទីផ្សារ ឬធ្វើពិពិធកម្ម។
  2. ចាត់ថ្នាក់ប្រភេទផលិតផលថ្មី (Classify New Product Types): វាយតម្លៃគំនិតផលិតផល ដោយផ្អែកលើចំណាត់ថ្នាក់ទាំង ៦ របស់ Booz-Allen & Hamilton ដើម្បីដឹងពីកម្រិតនៃភាពថ្មី (Degree of newness) ដែលតម្រូវឱ្យមានការរៀបចំថវិកា និងធនធានត្រឹមត្រូវ។
  3. រៀបចំរចនាសម្ព័ន្ធគ្រប់គ្រងក្រុមការងារ (Setup Management Structure): ប្រើប្រាស់ការគ្រប់គ្រងបែប Centralization សម្រាប់គម្រោងផលិតផលថ្មីស្រឡាង (Radical) និង Decentralization សម្រាប់ការកែលម្អផលិតផលតូចតាច (Incremental) ដើម្បីជៀសវាងជម្លោះផ្ទៃក្នុង។
  4. ស្រាវជ្រាវ និងប្រមូលទិន្នន័យអតិថិជន (Customer Data Collection): ធ្វើការស្រាវជ្រាវទីផ្សារ (Market Research) ដើម្បីប្រមូលមតិយោបល់ និងទិន្នន័យច្បាស់លាស់ពីអតិថិជនគោលដៅ មុននឹងសម្រេចចិត្តវិនិយោគផលិតក្នុងទ្រង់ទ្រាយធំ។

៥. វាក្យសព្ទបច្ចេកទេស (Technical Glossary)

ពាក្យបច្ចេកទេស (English) ការពន្យល់ជាខេមរភាសា (Khmer Explanation) និយមន័យសាមញ្ញ (Simple Definition)
new-to-the-world products ផលិតផលដែលបង្កើតឡើងថ្មីទាំងស្រុងដោយប្រើប្រាស់បច្ចេកវិទ្យាទំនើប និងមិនធ្លាប់មាននៅលើទីផ្សារពីមុនមក ដែលវាបង្កើតនូវទីផ្សារថ្មីមួយដាច់ដោយឡែក។ ដូចជាការបង្កើតទូរស័ព្ទស្មាតហ្វូនដំបូងគេបង្អស់នៅលើពិភពលោក ដែលផ្លាស់ប្តូររបៀបរស់នៅ និងការប្រាស្រ័យទាក់ទងរបស់មនុស្សទាំងស្រុង។
incremental new product ការកែលម្អ ឬបន្ថែមមុខងារថ្មីៗបន្តិចបន្តួចទៅលើផលិតផលដែលមានស្រាប់ ដោយប្រើប្រាស់បច្ចេកវិទ្យា ឬចំណេះដឹងដែលក្រុមហ៊ុនមានស្រាប់ដើម្បីបំពេញតម្រូវការអតិថិជនឱ្យកាន់តែប្រសើរ។ ដូចជាការបញ្ចេញទូរស័ព្ទស៊េរីថ្មីដែលមានកាមេរ៉ាច្បាស់ជាងមុន ឬថ្មកាន់បានយូរជាងស៊េរីឆ្នាំមុនបន្តិចបន្តួច។
concentric diversification យុទ្ធសាស្ត្រពង្រីកអាជីវកម្មដោយបង្កើតផលិតផល ឬសេវាកម្មថ្មីៗដែលមានទំនាក់ទំនង និងភាពស្រដៀងគ្នាទៅនឹងបច្ចេកវិទ្យា ឬទីផ្សារនៃផលិតផលដើមរបស់ក្រុមហ៊ុន។ ដូចជារោងចក្រផលិតទឹកក្រូចកំប៉ុង ងាកមកផលិតទឹកផ្លែឈើ ឬទឹកបរិសុទ្ធបន្ថែម ដោយប្រើប្រាស់ម៉ាស៊ីន និងបណ្តាញចែកចាយដែលខ្លួនមានស្រាប់។
