  • ការវាយតម្លៃអត្ថបទដោយអ្នកជំនាញ (Peer Review Process) ដែលមានអ្នកវាយតម្លៃចំនួន ៥៧ នាក់ សម្រាប់ពិនិត្យអត្ថបទទាំង ៣១។
  • ការបោះពុម្ព និងចែកចាយ (Printing and Distribution) ចំនួន ២ លេខ ដែលក្នុងមួយលេខមាន ៨៥០ ច្បាប់ សរុប ១៧០០ ច្បាប់។
  • ការតាមដានសន្ទស្សន៍ផលប៉ះពាល់ (Impact Factor Tracking) តាមរយៈមជ្ឈមណ្ឌលសន្ទស្សន៍អំណះអំណាងទស្សនាវដ្តីថៃ (TCI)។

សេចក្តីសន្និដ្ឋានសំខាន់ៗ (Key Conclusions)៖

  • គម្រោងនេះសម្រេចបានការបោះពុម្ពទស្សនាវដ្តីចំនួន ២ លេខ (ខែមីនា និងកញ្ញា ឆ្នាំ២០១៥) ដែលមានអត្ថបទសរុបចំនួន ៣១ អត្ថបទ (អត្ថបទស្រាវជ្រាវ ២៨ អត្ថបទសិក្សា ២ និងអត្ថបទពិសេស ១)។
  • ទស្សនាវដ្តីសរុប ១៧០០ ច្បាប់ ត្រូវបានចែកចាយទៅកាន់ទីភ្នាក់ងារចំនួន ៤៤៤ រួមមានស្ថាប័នរដ្ឋ ឯកជន បណ្ណាល័យជាតិ និងសាកលវិទ្យាល័យនានា ព្រមទាំងមានសន្ទស្សន៍ផលប៉ះពាល់ជាមធ្យម ០,០៦៧។

២. ការរកឃើញសំខាន់ៗ (Key Findings)

របាយការណ៍នេះបង្ហាញពីភាពជោគជ័យនៃការបោះពុម្ពទស្សនាវដ្តីស្រាវជ្រាវរបស់សាកលវិទ្យាល័យ RMUTP ចំនួន ២ លេខ ក្នុងឆ្នាំ២០១៥ ដែលមានអត្ថបទសរុប ៣១ និងចែកចាយដល់ស្ថាប័នចំនួន ៤៤៤។ គម្រោងនេះបានប្រើប្រាស់ថវិកាប្រកបដោយប្រសិទ្ធភាព និងសម្រេចបានសន្ទស្សន៍ផលប៉ះពាល់ (Impact Factor) គួរឱ្យកត់សម្គាល់នៅក្នុងមជ្ឈមណ្ឌល TCI។

ការរកឃើញ (Finding) ព័ត៌មានលម្អិត (Detail) ភស្តុតាង (Evidence)
ការបោះពុម្ព និងការចែកចាយ (Publication and Distribution) ទស្សនាវដ្តីត្រូវបានបោះពុម្ពចំនួន ២ លេខ (ខែមីនា និងកញ្ញា) សរុប ១៧០០ ច្បាប់ និងចែកចាយដល់ស្ថាប័នរដ្ឋ ឯកជន និងបណ្ណាល័យចំនួន ៤៤៤ ដើម្បីផ្សព្វផ្សាយស្នាដៃ។ ទិន្នន័យបង្ហាញថាមានអត្ថបទចំនួន ៣១ ត្រូវបានបោះពុម្ព និងមានទស្សនាវដ្តី ៨៥០ ច្បាប់ក្នុងមួយលេខត្រូវបានចែកចាយទៅកាន់បណ្តាញចំនួន ៨៩ និងស្ថាប័នក្រៅ ៤៤៤ កន្លែង។
ការគ្រប់គ្រងថវិកាប្រកបដោយប្រសិទ្ធភាព (Efficient Budget Management) ការចំណាយជាក់ស្តែងសម្រាប់ការបោះពុម្ពមានចំនួនតិចជាងថវិកាដែលបានអនុម័ត ដោយបង្ហាញពីការសន្សំសំចៃ ខណៈដែលការចំណាយលើអ្នកវាយតម្លៃអត្ថបទត្រូវបានប្រើប្រាស់ពីប្រភពចំណូលផ្សេង។ ថវិកាអនុម័តមានចំនួន ២៥០.០០០ បាត ប៉ុន្តែចំណាយជាក់ស្តែងត្រឹមតែ ២១៩.៩៤៩ បាត។ កម្រៃសេវាអ្នកវាយតម្លៃអត្ថបទ (Peer Reviewers) ចំនួន ៨២.២០០ បាត ត្រូវបានប្រើប្រាស់ពីប្រភពចំណូលរបស់សាកលវិទ្យាល័យដោយផ្ទាល់។
សន្ទស្សន៍ផលប៉ះពាល់ និងគុណភាព (Impact Factor and Quality Assurance) ទស្សនាវដ្តីនេះទទួលបានការទទួលស្គាល់គុណភាពពីមជ្ឈមណ្ឌលសន្ទស្សន៍អំណះអំណាងទស្សនាវដ្តីថៃ (TCI) និងមានសន្ទស្សន៍ផលប៉ះពាល់ដែលអាចប្រកួតប្រជែងបានក្នុងចំណោមសាកលវិទ្យាល័យបច្ចេកវិទ្យា។ ទស្សនាវដ្តីត្រូវបានបញ្ជាក់គុណភាពដោយ TCI សម្រាប់រយៈពេល ៥ឆ្នាំ (រហូតដល់ឆ្នាំ២០១៩) និងទទួលបាន Impact Factor មធ្យម (ឆ្នាំ២០១២-២០១៤) ស្មើនឹង ០,០៦៧។

ដោយផ្អែកលើរបាយការណ៍នេះ សាកលវិទ្យាល័យ និងស្ថាប័នស្រាវជ្រាវគួរតែពង្រឹងយន្តការបោះពុម្ពផ្សាយ និងប្រព័ន្ធវាយតម្លៃគុណភាពអត្ថបទដើម្បីឈានទៅរកស្តង់ដារអន្តរជាតិ។

របាយការណ៍នេះផ្តល់នូវគំរូដ៏ល្អសម្រាប់គ្រឹះស្ថានឧត្តមសិក្សានៅកម្ពុជា ក្នុងការរៀបចំ គ្រប់គ្រង និងវាយតម្លៃទស្សនាវដ្តីស្រាវជ្រាវរបស់ខ្លួន។ ការយល់ដឹងពីដំណើរការវាយតម្លៃអត្ថបទ (Peer review) ការគ្រប់គ្រងថវិកាបោះពុម្ព និងយុទ្ធសាស្ត្រក្នុងការចូលទៅកាន់សន្ទស្សន៍អំណះអំណាងអាស៊ាន (ACI) គឺជាភាពចាំបាច់សម្រាប់កម្ពុជាក្នុងការលើកកម្ពស់គុណភាពអប់រំកម្រិតឧត្តមសិក្សា។

ផលប៉ះពាល់មូលដ្ឋាន (Local Implications)៖

  • វិស័យឧត្តមសិក្សា (Higher Education Sector): សាកលវិទ្យាល័យនៅកម្ពុជា (ឧ. សាកលវិទ្យាល័យភូមិន្ទភ្នំពេញ វិទ្យាស្ថានបច្ចេកវិទ្យាកម្ពុជា) អាចយកគំរូនៃការបែងចែកថវិកា និងយន្តការចែករំលែកស្នាដៃស្រាវជ្រាវនេះ ទៅកសាងទស្សនាវដ្តីវិទ្យាសាស្ត្រក្នុងស្រុកឱ្យមានស្តង់ដារខ្ពស់។
  • គោលនយោបាយស្រាវជ្រាវ (Research Policy): ក្រសួងអប់រំ យុវជន និងកីឡា (MoEYS) អាចប្រើប្រាស់ទិន្នន័យ និងនីតិវិធីនៃការគ្រប់គ្រងនេះ ដើម្បីជំរុញឱ្យមានការបង្កើតមជ្ឈមណ្ឌលសន្ទស្សន៍អំណះអំណាងកម្ពុជា ឬគាំទ្រទស្សនាវដ្តីក្នុងស្រុកឱ្យចូលក្នុងប្រព័ន្ធ ASEAN Citation Index (ACI) ។
  • បណ្តាញអ្នកជំនាញ (Professional Networks): ការកសាងបណ្តាញអ្នកវាយតម្លៃអត្ថបទ (Peer Reviewer Pool) ដែលមានអ្នកជំនាញឆ្លងស្ថាប័ន នឹងជួយលើកកម្ពស់តម្លាភាព និងទំនុកចិត្តលើការស្រាវជ្រាវរបស់កម្ពុជា។

ការអនុវត្តតាមស្តង់ដារនៃការគ្រប់គ្រងទស្សនាវដ្តីវិទ្យាសាស្ត្រតាមបែបផែននេះ នឹងជួយឱ្យសាកលវិទ្យាល័យនៅកម្ពុជាអាចពង្រីកឥទ្ធិពលនៃការស្រាវជ្រាវរបស់ខ្លួន ទាំងក្នុងឆាកជាតិ និងតំបន់អាស៊ាន ប្រកបដោយនិរន្តរភាព និងតម្លាភាពហិរញ្ញវត្ថុ។

៥. ផែនការអនុវត្ត (Implementation Roadmap)

ដើម្បីអនុវត្តតាមអនុសាសន៍នៃរបាយការណ៍នេះ គួរអនុវត្តតាមជំហានខាងក្រោម៖

  1. ការរៀបចំរចនាសម្ព័ន្ធ និងថវិកា (Establishment of Structure and Budget): បង្កើតគណៈកម្មការនិពន្ធច្បាស់លាស់ និងស្នើសុំកញ្ចប់ថវិកាប្រចាំឆ្នាំ សម្រាប់ការបោះពុម្ព និងការបើកប្រាក់កម្រៃដល់អ្នកវាយតម្លៃអត្ថបទ (Peer Reviewers) ដោយយោងតាមលិខិតបទដ្ឋានរបស់សាកលវិទ្យាល័យ។
  2. ការកសាងបណ្តាញអ្នកវាយតម្លៃ និងប្រមូលអត្ថបទ (Building Reviewer Network and Call for Papers): អញ្ជើញអ្នកជំនាញទាំងក្នុង និងក្រៅស្ថាប័ន ដើម្បីបង្កើតជាបណ្តាញអ្នកវាយតម្លៃ (Reviewer Pool) និងធ្វើការផ្សព្វផ្សាយដើម្បីទទួលអត្ថបទស្រាវជ្រាវ (Call for Papers) ពីអ្នកស្រាវជ្រាវ។
  3. ការពង្រឹងដំណើរការវាយតម្លៃ (Strengthening Peer Review Process): អនុវត្តដំណើរការវាយតម្លៃយ៉ាងតឹងរ៉ឹង (Double-blind peer review) និងត្រូវមានការតាមដានរហ័ស ដោយផ្តល់មតិកែលម្អត្រឡប់ទៅអ្នកនិពន្ធក្នុងរយៈពេលកំណត់ (ឧ. ១ខែ) ដើម្បីធានាគុណភាពអត្ថបទ។
  4. ការបោះពុម្ព និងការចែកចាយ (Publication and Wide Distribution): បោះពុម្ពទស្សនាវដ្តី (ទាំងទម្រង់សៀវភៅបោះពុម្ព និងទម្រង់ឌីជីថល e-Journal) រួចចែកចាយទៅកាន់បណ្ណាល័យ ស្ថាប័នរដ្ឋ ឯកជន និងសាកលវិទ្យាល័យដៃគូផ្សេងៗ ដើម្បីបង្កើនការអាន។
  5. ការតាមដានសន្ទស្សន៍ និងការវាយតម្លៃ (Monitoring and Impact Evaluation): ដាក់បញ្ចូលទស្សនាវដ្តីទៅក្នុងមូលដ្ឋានទិន្នន័យអន្តរជាតិ ឬថ្នាក់តំបន់ (ដូចជា ACI) ដើម្បីតាមដានសន្ទស្សន៍អំណះអំណាង (Citation Index) គណនា Impact Factor និងរៀបចំរបាយការណ៍វាយតម្លៃប្រចាំឆ្នាំជូនថ្នាក់ដឹកនាំ។

៦. វាក្យសព្ទបច្ចេកទេស (Technical Glossary)

ពាក្យបច្ចេកទេស (English) ការពន្យល់ជាខេមរភាសា (Khmer Explanation) និយមន័យសាមញ្ញ (Simple Definition)
TCI (Thai-Journal Citation Index) មជ្ឈមណ្ឌលផ្ទុកទិន្នន័យ និងវាយតម្លៃគុណភាពទស្សនាវដ្តីសិក្សាស្រាវជ្រាវរបស់ប្រទេសថៃ ដែលគណនាសន្ទស្សន៍នៃការដកស្រង់អត្ថបទដើម្បីធានាស្តង់ដារបោះពុម្ពផ្សាយ និងលើកកម្ពស់គុណភាពអប់រំកម្រិតឧត្តមសិក្សា។ ដូចជាបញ្ជីចំណាត់ថ្នាក់ស្តង់ដារប្រចាំប្រទេស ដែលវាយតម្លៃថាទស្សនាវដ្តីមួយណាមានគុណភាព និងមានគេយកទៅប្រើប្រាស់ច្រើនជាងគេ។
Impact Factor សន្ទស្សន៍ដែលវាស់វែងពីឥទ្ធិពល និងគុណភាពនៃទស្សនាវដ្តីមួយ ដោយគណនាពីចំនួនមធ្យមភាគនៃការដកស្រង់ (Citation) នៃអត្ថបទដែលបានបោះពុម្ពក្នុងទស្សនាវដ្តីនោះ ក្នុងរយៈពេលជាក់លាក់ណាមួយ (ជាទូទៅគិតជាប្រចាំឆ្នាំ)។ ដូចជារង្វាស់ប្រជាប្រិយភាពនិងឥទ្ធិពល បើមានអ្នកស្រាវជ្រាវផ្សេងយកអត្ថបទនោះទៅយោងច្រើន មានន័យថាទស្សនាវដ្តីនោះមានតម្លៃនិងពិន្ទុខ្ពស់។
Peer Review ដំណើរការវាយតម្លៃអត្ថបទដោយអ្នកជំនាញ (Reviewers) ដែលមានឯកទេសស្រដៀងគ្នា ដើម្បីធានាថាអត្ថបទស្រាវជ្រាវនោះមានគុណភាព ភាពត្រឹមត្រូវ និងគួរឱ្យទុកចិត្តមុននឹងសម្រេចចិត្តបោះពុម្ពផ្សាយ។ ដូចជាការត្រួតពិនិត្យស្នាដៃដោយគណៈកម្មការជំនាញដ៏តឹងរ៉ឹង មុននឹងអនុញ្ញាតឱ្យស្នាដៃនោះចេញផ្សាយជាផ្លូវការ។
Research Article អត្ថបទវិទ្យាសាស្ត្រដែលសរសេរឡើងដោយផ្អែកលើការស្រាវជ្រាវថ្មីៗ ដោយមានការកំណត់បញ្ហា ការប្រមូលទិន្នន័យ ការវិភាគ និងការរកឃើញលទ្ធផលថ្មី (Original findings) ដែលមិនទាន់មានអ្នករកឃើញពីមុនមក។ ដូចជារបាយការណ៍លម្អិតពីការរកឃើញរបកគំហើញថ្មីមួយ បន្ទាប់ពីបានចុះធ្វើការពិសោធន៍ ឬប្រមូលទិន្នន័យជាក់ស្តែងរួចរាល់។
Academic Article អត្ថបទដែលសរសេរឡើងដើម្បីវិភាគ សំយោគ ឬពន្យល់ពីទ្រឹស្តី និងចំណេះដឹងមានស្រាប់ ដើម្បីផ្តល់នូវទស្សនៈ ការវាយតម្លៃ ឬការយល់ដឹងថ្មីលើប្រធានបទណាមួយ ដោយមិនចាំបាច់មានទិន្នន័យពិសោធន៍ថ្មីនោះទេ។ ដូចជាការសរសេរអត្ថបទវិភាគនិងបូកសរុបមេរៀនចាស់ៗ ដើម្បីឱ្យអ្នកអានងាយយល់និងឃើញពីទិដ្ឋភាពទូទៅនៃបញ្ហាមួយយ៉ាងច្បាស់លាស់។
ASEAN Citation Index (ACI) មូលដ្ឋានទិន្នន័យកម្រិតតំបន់ដែលប្រមូលផ្តុំ និងតាមដានការដកស្រង់អត្ថបទស្រាវជ្រាវពីបណ្តាប្រទេសក្នុងតំបន់អាស៊ាន (ASEAN) ដើម្បីលើកកម្ពស់គុណភាពទស្សនាវដ្តីតំបន់ឱ្យឈានទៅរកស្តង់ដារអន្តរជាតិ (ដូចជា Scopus ឬ Web of Science)។ ដូចជាឆាកប្រកួតប្រជែងគុណភាពស្នាដៃស្រាវជ្រាវកម្រិតអាស៊ាន ដែលរៀបចំឡើងដើម្បីពង្រឹងសមត្ថភាពមុននឹងឈានទៅប្រកួតលើឆាកអន្តរជាតិ។
