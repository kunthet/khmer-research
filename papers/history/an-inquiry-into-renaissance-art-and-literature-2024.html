  • ការវិភាគលើលក្ខណៈនិងបច្ចេកទេសសិល្បៈ (Analysis of Artistic Characteristics) ដូចជា ការប្រើប្រាស់ពន្លឺនិងស្រមោល (Chiaroscuro) និងការគូរមានជម្រៅ (Perspective)

  • ទស្សនវិជ្ជាមនុស្សធម៌ (Humanism) បានផ្លាស់ប្តូរការផ្តោតអារម្មណ៍យ៉ាងខ្លាំងពីគំនាបសាសនានៃយុគសម័យកណ្តាល មកការលើកកម្ពស់សេចក្តីថ្លៃថ្នូរ តម្លៃ និងហេតុផលរបស់មនុស្សពិតប្រាកដ។
  • នវានុវត្តន៍នៃបច្ចេកទេសសិល្បៈ ដូចជាការសិក្សាកាយវិភាគសាស្ត្រ (Human Anatomy) និងការប្រើប្រាស់ពន្លឺនិងស្រមោល (Chiaroscuro) របស់វិចិត្រករល្បីៗ បានក្លាយជាមូលដ្ឋានគ្រឹះដ៏សំខាន់សម្រាប់សិល្បៈ ភាពយន្ត និងការថតរូបសម័យទំនើប។
  • ការធ្វើសមាហរណកម្មយ៉ាងជិតស្និទ្ធរវាងសិល្បៈ និងអក្សរសិល្ប៍ បានជំរុញយ៉ាងខ្លាំងដល់ការរំដោះបញ្ញាស្មារតី និងផ្តល់ជាឯកសារយោងដ៏សំខាន់សម្រាប់ការបង្កើតអក្សរសិល្ប៍ និងគំនិតនយោបាយនាសម័យទំនើប។

Medieval Art & Literature (Baseline)

Renaissance Art Techniques (Perspective, Anatomy, Chiaroscuro)
បច្ចេកទេសសិល្បៈសម័យរស់ឡើងវិញ (គំនូរមានជម្រៅ កាយវិភាគសាស្ត្រ និងការប្រើពន្លឺនិងស្រមោល)		 បង្កើតបានជាផ្ទាំងគំនូររស់រវើក មានលក្ខណៈធម្មជាតិ និងអាចបង្ហាញពីលំហបីវិមាត្រ (3D) លើផ្ទៃរាបស្មើបានយ៉ាងល្អ។ ទាមទារឱ្យសិល្បករមានចំណេះដឹងស៊ីជម្រៅផ្នែកគណិតវិទ្យា កាយវិភាគសាស្ត្រ និងរូបវិទ្យានៃពន្លឺ។ ផ្តល់ជាមូលដ្ឋានគ្រឹះដ៏រឹងមាំសម្រាប់សិល្បៈទំនើប ការថតរូប និងវិស័យភាពយន្តនាសម័យកាលបច្ចុប្បន្ន។
Humanist Literature
អក្សរសិល្ប៍បែបមនុស្សធម៌ (Humanism)		 លើកកម្ពស់តម្លៃនិងសេចក្តីថ្លៃថ្នូររបស់មនុស្ស និងហ៊ានបង្ហាញពីការពិតនៃសង្គម នយោបាយ និងអារម្មណ៍ពិតរបស់បុគ្គល។ ប្រឈមមុខនឹងការរិះគន់ គំនាប និងការចាប់ដាក់គុកពីសំណាក់ស្ថាប័នសាសនា ឬអ្នកកាន់អំណាចផ្តាច់ការនៅសម័យនោះ។ បានបំបែកទម្លាប់នៃការគ្រប់គ្រងខាងទ្រឹស្តីសាសនា និងបានត្រួសត្រាយផ្លូវសម្រាប់ការអភិវឌ្ឍគំនិតនយោបាយ និងអក្សរសាស្ត្រទំនើប។

ការចំណាយលើធនធាន (Resource Cost)៖ ឯកសារបានបញ្ជាក់ថាការរីកចម្រើននៃសិល្បៈ និងអក្សរសិល្ប៍នាសម័យនោះ ទាមទារឱ្យមានការគាំទ្រផ្នែកហិរញ្ញវត្ថុ ការស្រាវជ្រាវ និងការច្នៃប្រឌិតខ្ពស់។

  • Funding and Patronage: ទាមទារការឧបត្ថម្ភគាំទ្រពីគ្រួសារអភិជន (ដូចជាត្រកូល Medici) ឬស្ថាប័នព្រះវិហារ ដើម្បីផ្គត់ផ្គង់ការរស់នៅ និងការធ្វើដំណើររបស់សិល្បករ។
  • Expertise (Anatomy & Classical Literature): សិល្បករនិងអ្នកនិពន្ធត្រូវចំណាយពេលសិក្សាស៊ីជម្រៅលើអត្ថបទបុរាណក្រិក-រ៉ូម៉ាំង និងរចនាសម្ព័ន្ធកាយវិភាគសាស្ត្ររបស់មនុស្ស។
  • Hardware/Technology (Printmaking): ត្រូវការបច្ចេកវិទ្យាបោះពុម្ព (Printmaking) ដែលកំពុងរីកដុះដាលនៅទីក្រុង Venice ដើម្បីផ្សព្វផ្សាយស្នាដៃគំនូរ និងសៀវភៅឱ្យបានទូលំទូលាយ។

ភាពលំអៀងនៃទិន្នន័យ (Data Bias)៖

ការសិក្សានេះផ្តោតទាំងស្រុងទៅលើបរិបទប្រវត្តិសាស្ត្រសង្គម និងវប្បធម៌របស់ទ្វីបអឺរ៉ុប (ជាពិសេសប្រទេសអ៊ីតាលី និងអង់គ្លេស) ចន្លោះសតវត្សទី១៤ ដល់ទី១៧ ដោយមិនមានការរៀបរាប់ពីទ្វីបផ្សេងឡើយ។ ទោះជាយ៉ាងណាក៏ដោយ សម្រាប់ប្រទេសកម្ពុជា ការយល់ដឹងពីការផ្លាស់ប្តូរផ្នត់គំនិតពីការតឹងរ៉ឹងផ្នែកសាសនាមកកាន់ការលើកកម្ពស់តម្លៃមនុស្ស (Humanism) គឺមានសារៈសំខាន់សម្រាប់ការស្តារ និងអភិវឌ្ឍវប្បធម៌ទំនើប។

លទ្ធភាពនៃការអនុវត្ត (Applicability)៖

ទោះបីជាឯកសារនេះនិយាយពីប្រវត្តិសាស្ត្រអឺរ៉ុបក៏ដោយ គំនិតនៃការបង្កើតយុគសម័យរស់ឡើងវិញ (Renaissance) គឺមានប្រយោជន៍ខ្លាំងសម្រាប់ការអភិវឌ្ឍវិស័យសិល្បៈ និងអក្សរសាស្ត្រនៅកម្ពុជា។

  • សាកលវិទ្យាល័យភូមិន្ទវិចិត្រសិល្បៈ (Royal University of Fine Arts): អាចបញ្ចូលការសិក្សាស៊ីជម្រៅលើបច្ចេកទេសគូរគំនូរដោយប្រើពន្លឺនិងស្រមោល (Chiaroscuro) និងកាយវិភាគសាស្ត្រ ដើម្បីយកមកអភិវឌ្ឍគំនូរបុរាណខ្មែរឱ្យកាន់តែមានជីវិត និងទំនើបកម្ម។
  • វិស័យអក្សរសិល្ប៍ និងភាពយន្តខ្មែរ (Cambodian Literature & Cinema): អ្នកនិពន្ធនិងផលិតករភាពយន្តអាចរៀនសូត្រពីទស្សនវិជ្ជាមនុស្សធម៌ ដើម្បីផលិតស្នាដៃដែលឆ្លុះបញ្ចាំងពីជីវិតពិត ការតស៊ូ និងសេចក្តីថ្លៃថ្នូររបស់ប្រជាជនកម្ពុជាទូទៅ ជាជាងការផ្តោតតែលើរឿងព្រេងទេវកថា។
  • ការឧបត្ថម្ភសិល្បៈដោយវិស័យឯកជន (Corporate Art Patronage): អាចយកគំរូតាមត្រកូលអភិជននៅអ៊ីតាលី ដោយលើកទឹកចិត្តដល់ក្រុមហ៊ុនធំៗនៅកម្ពុជា ឱ្យបង្កើតមូលនិធិគាំទ្រវិចិត្រករ និងអ្នកនិពន្ធវ័យក្មេង ដើម្បីជំរុញនវានុវត្តន៍វប្បធម៌ជាតិ។

ការបណ្តុះស្មារតីច្នៃប្រឌិត និងការគាំទ្រសិល្បករដោយផ្អែកលើតម្លៃមនុស្សធម៌ អាចជាកាតាលីករជំរុញឱ្យមានការរស់ឡើងវិញនៃសិល្បៈ និងវប្បធម៌ (Cambodian Renaissance) ប្រកបដោយភាពរឹងមាំ និងទំនើប។

៤. ផែនការសកម្មភាពសម្រាប់និស្សិត (Actionable Roadmap)

ដើម្បីអនុវត្តតាមការសិក្សានេះ និស្សិតគួរអនុវត្តតាមជំហានខាងក្រោម៖

  1. សិក្សាពីបច្ចេកទេសមូលដ្ឋាននៃសិល្បៈទំនើប: និស្សិតផ្នែកសិល្បៈត្រូវអនុវត្តការប្រើប្រាស់ពន្លឺនិងស្រមោល (Chiaroscuro) និងគំនូរមានជម្រៅ (Linear Perspective) ទាំងនៅក្នុងគំនូរផ្ទាល់ និងតាមរយៈកម្មវិធីឌីជីថលដូចជា Adobe Photoshop ឬ Procreate។
  2. ស្រាវជ្រាវ និងអនុវត្តកាយវិភាគសាស្ត្រសិល្បៈ: ធ្វើការសិក្សាលម្អិតពីរចនាសម្ព័ន្ធរាងកាយមនុស្ស (Human Anatomy) ដើម្បីអាចគូររូបភាពតួអង្គខ្មែរ (ឧទាហរណ៍៖ តួអង្គក្នុងរបាំព្រះរាជទ្រព្យ) ឱ្យមានលក្ខណៈរស់រវើក ត្រឹមត្រូវតាមសមាមាត្រ និងបង្ហាញពីចលនាសាច់ដុំបានច្បាស់លាស់។
  3. វិភាគស្នាដៃអក្សរសិល្ប៍បែបមនុស្សធម៌: និស្សិតអក្សរសាស្ត្រគួរអាន និងប្រៀបធៀបស្នាដៃលោកខាងលិច (ឧ. សៀវភៅ The Prince ឬ Decameron) ជាមួយនឹងអក្សរសិល្ប៍ខ្មែរ ដើម្បីរៀនពីរបៀបសរសេររៀបរាប់ពីអារម្មណ៍ និងចិត្តសាស្ត្ររបស់មនុស្សពិតប្រាកដក្នុងសង្គម។
  4. បង្កើតកិច្ចសហការអន្តរវិស័យរវាងសិល្បៈ និងអក្សរសាស្ត្រ: រៀបចំក្រុមការងារដែលរួមមានវិចិត្រករ និងអ្នកនិពន្ធ ដើម្បីបង្កើតគម្រោងរួមគ្នា (ឧទាហរណ៍៖ ការធ្វើសៀវភៅគំនូររឿង ឬ Graphic Novels) ដោយប្រើប្រាស់ Adobe InDesign ក្នុងការរចនា និងបោះពុម្ពផ្សាយ។
  5. ស្វែងរកការគាំទ្រ និងតាំងពិព័រណ៍ស្នាដៃ: ចងក្រងស្នាដៃសិល្បៈដែលឆ្លុះបញ្ចាំងពីតម្លៃមនុស្សធម៌ និងបញ្ហាសង្គម រួចយកទៅតាំងបង្ហាញនៅតាមវិចិត្រសាលនានា ដូចជា Bophana Center ឬ Factory Phnom Penh ដើម្បីស្វែងរកការគាំទ្រពីសាធារណជន និងអ្នកវិនិយោគ។

៥. វាក្យសព្ទបច្ចេកទេស (Technical Glossary)

ពាក្យបច្ចេកទេស (English) ការពន្យល់ជាខេមរភាសា (Khmer Explanation) និយមន័យសាមញ្ញ (Simple Definition)
Renaissance គឺជាចលនាវប្បធម៌ និងប្រវត្តិសាស្ត្រនៅអឺរ៉ុប (សតវត្សទី១៤ ដល់ទី១៧) ដែលផ្តោតលើការរស់ឡើងវិញ ឬការនាំយកមកវិញនូវចំណេះដឹងទស្សនវិជ្ជា សិល្បៈ និងអក្សរសាស្ត្រកាលពីសម័យក្រិក និងរ៉ូមបុរាណ ដើម្បីផ្លាស់ប្តូរ និងរំដោះសង្គមចេញពីគំនាបសាសនានាសម័យកណ្តាល។ ដូចជាការក្រោកពីគេងបន្ទាប់ពីយប់ដ៏ងងឹត (យុគសម័យកណ្តាល) ដើម្បីចាប់ផ្តើមថ្ងៃថ្មីដែលពោរពេញដោយពន្លឺនៃចំណេះដឹង ការច្នៃប្រឌិត និងការគិតបែបវិទ្យាសាស្ត្រ។
Humanism ជាទស្សនវិជ្ជាស្នូលនៃសម័យរស់ឡើងវិញ ដែលផ្លាស់ប្តូរការផ្តោតអារម្មណ៍ពីការបម្រើព្រះនិងសាសនាទាំងស្រុង មកផ្តល់តម្លៃលើសមត្ថភាព សេចក្តីថ្លៃថ្នូរ និងហេតុផលរបស់មនុស្សជាតិ ព្រមទាំងលើកទឹកចិត្តឲ្យបុគ្គលស្វែងរកចំណេះដឹងដើម្បីអភិវឌ្ឍខ្លួន។ ដូចជាការជឿជាក់ថា "មនុស្សគឺជាអ្នកកំណត់វាសនារបស់ខ្លួនឯងតាមរយៈចំណេះដឹង និងការប្រឹងប្រែង" ជាជាងការទន្ទឹងរង់ចាំតែការជួយសង្គ្រោះពីទេវតា ឬព្រះ។
chiaroscuro គឺជាបច្ចេកទេសសិល្បៈក្នុងការគូរគំនូរ ដោយប្រើប្រាស់ភាពផ្ទុយគ្នាយ៉ាងខ្លាំងរវាងពន្លឺ និងស្រមោល (Light and Dark) ដើម្បីបង្កើតជម្រៅ និងសាច់មាឌ ធ្វើឱ្យរូបភាពមានសភាពលេចធ្លោចេញពីផ្ទាំងគំនូររាបស្មើ និងមើលទៅមានជីវិតរស់រវើក (លក្ខណៈ 3D)។ ដូចជាការបញ្ចាំងភ្លើងពិលទៅលើមុខមនុស្សក្នុងបន្ទប់ងងឹត ដែលធ្វើឱ្យយើងឃើញខ្ទង់ច្រមុះ និងទម្រង់មុខលេចចេញមកយ៉ាងច្បាស់។
linear perspective ជាបច្ចេកទេសគូរគំនូរដោយប្រើខ្សែបន្ទាត់ និងចំណុចរួម (vanishing point) ដើម្បីបង្កើតការបំភាន់ភ្នែកឱ្យឃើញមានជម្រៅ និងលំហបីវិមាត្រនៅលើផ្ទៃរាបស្មើ (2D) ដែលជួយកំណត់ទីតាំង និងទំហំវត្ថុទៅតាមចម្ងាយជិតឬឆ្ងាយ។ ដូចជាពេលយើងឈរមើលផ្លូវរថភ្លើងដ៏វែងអន្លាយ ដែលផ្លូវដែកទាំងពីរមើលទៅដូចជាតូចស្រួចរួមគ្នាទៅកាន់ចំណុចមួយនៅឯជើងមេឃ។
secularism គឺជាការផ្លាស់ប្តូរផ្នត់គំនិតចេញពីការគ្រប់គ្រងនិងឥទ្ធិពលរបស់ស្ថាប័នសាសនា ដោយងាកមកចាប់អារម្មណ៍លើជីវិតរស់នៅជាក់ស្តែង ទិដ្ឋភាពសង្គម ធម្មជាតិ និងតម្រូវការពិតប្រាកដរបស់មនុស្សលោកនៅក្នុងពិភពលោកបច្ចុប្បន្ន។ ដូចជាការប្តូរប្រធានបទសន្ទនាពីការនិយាយរឿងឋានសួគ៌និងនរក មកនិយាយពីការប្រកបរបរចិញ្ចឹមជីវិត ស្នេហា និងភាពសប្បាយរីករាយក្នុងពេលបច្ចុប្បន្ន។
frescoes គឺជាបច្ចេកទេសគូរគំនូរដោយប្រើទឹកពណ៌ផាត់ទៅលើជញ្ជាំងកំបោរសើម (plaster) ដើម្បីឱ្យពណ៌ជ្រាបចូលទៅក្នុងសាច់កំបោរពេលវាស្ងួត ធ្វើឱ្យផ្ទាំងគំនូរជាប់បានយូរអង្វែង ដែលគេនិយមប្រើសម្រាប់ការគូរតុបតែងដំបូល ឬជញ្ជាំងព្រះវិហារ។ ដូចជាការសាក់រូបលើស្បែកមនុស្ស ដែលពណ៌ជាប់ជ្រាបចូលទៅក្នុងសាច់ ជាជាងការគ្រាន់តែគូរលាបពីលើស្បែកដែលអាចលុបជ្រះបានងាយស្រួល។
