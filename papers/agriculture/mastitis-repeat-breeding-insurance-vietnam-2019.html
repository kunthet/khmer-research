  • តំបន់ចិញ្ចឹមគោយកទឹកដោះពាណិជ្ជកម្ម (ឧ. ស្រុកបាទី ខេត្តតាកែវ): កសិដ្ឋានធំៗដូចជាកសិដ្ឋានគិរីសួគ៌ អាចអនុវត្តការធ្វើតេស្តសំណាកទឹកដោះគោជាប្រចាំ ដើម្បីតាមដានជំងឺរលាកសុដន់លាក់រោគសញ្ញា (Sub-clinical mastitis) ជៀសវាងការធ្លាក់ចុះទិន្នផលទឹកដោះ។
  • អគ្គនាយកដ្ឋានសុខភាពសត្វ និងផលិតកម្មសត្វ (MAFF): អាចប្រើប្រាស់លទ្ធផលនេះដើម្បីយល់ថា កសិករប្រហែលជាមិនសម្រេចចិត្តទិញធានារ៉ាប់រងដោយសារតែជំងឺរ៉ាំរ៉ៃនោះទេ ដូច្នេះការរចនាគោលនយោបាយធានារ៉ាប់រងគួរតែបន្ថែមការគាំទ្រដល់ការវាយតម្លៃអន្តរាគមន៍កត្តាសង្គម និងការយល់ដឹង។

អ្នករៀបចំគោលនយោបាយនៅកម្ពុជាគួរកត់សម្គាល់ថា កត្តាសង្គមវិទ្យា និងការប្រើប្រាស់បច្ចេកទេសគ្រប់គ្រងកសិដ្ឋាន (ឧ. ម៉ាស៊ីនរឹតទឹកដោះ) មានឥទ្ធិពលលើការសម្រេចចិត្តទិញធានារ៉ាប់រង ច្រើនជាងអត្រានៃការកើតជំងឺរ៉ាំរ៉ៃ។

៤. ផែនការសកម្មភាពសម្រាប់និស្សិត (Actionable Roadmap)

ដើម្បីអនុវត្តតាមការសិក្សានេះ និស្សិតគួរអនុវត្តតាមជំហានខាងក្រោម៖

  1. រៀនពីការរចនាការស្ទង់មតិ និងការប្រមូលទិន្នន័យ: ចាប់ផ្តើមរៀនពីរបៀបបង្កើតកម្រងសំណួរ (Questionnaire) និងការគណនាទំហំសំណាក (Sample size calculation) ដោយប្រើប្រាស់ឧបករណ៍ដូចជា Epi Info ឬ KoboToolbox សម្រាប់ការស្ទង់មតិនៅតាមកសិដ្ឋាន។
  2. អនុវត្តការវិភាគទិន្នន័យស្ថិតិបសុពេទ្យ: ដំឡើងកម្មវិធី R Statistical Software ហើយរៀនសរសេរកូដដើម្បីវិភាគទំនាក់ទំនងរវាងអថេរ ដូចជាការអនុវត្តតេស្ត Wilcoxon rank-sum និង Chi-square លើទិន្នន័យអត្រាការកើតជំងឺ។
  3. ស្វែងយល់ពីបច្ចេកទេសមន្ទីរពិសោធន៍: សិក្សាពីនីតិវិធីនៃការបណ្តុះមេរោគលើបន្ទះ Sheep blood agar និងការប្រើប្រាស់ម៉ាស៊ីន PCR ព្រមទាំងការធ្វើតេស្តជីវគីមី ដើម្បីកំណត់អត្តសញ្ញាណបាក់តេរី Staphylococcus aureus។
  4. វិភាគពីអាកប្បកិរិយាសេដ្ឋកិច្ចក្នុងកសិកម្ម: អានស្រាវជ្រាវអំពីសេដ្ឋកិច្ចអាកប្បកិរិយា (Behavioral Economics) ដោយប្រើប្រាស់ Risk Analysis Frameworks ដើម្បីយល់ពីកត្តាសង្គមវិទ្យាដែលជំរុញឱ្យកសិករកម្ពុជាសម្រេចចិត្តចូលរួមក្នុងកម្មវិធីធានារ៉ាប់រងកសិកម្ម។

៥. វាក្យសព្ទបច្ចេកទេស (Technical Glossary)

ពាក្យបច្ចេកទេស ការពន្យល់ជាខេមរភាសា (Khmer Explanation) និយមន័យសាមញ្ញ (Simple Definition)
Sub-clinical mastitis (ជំងឺរលាកសុដន់មិនបញ្ចេញរោគសញ្ញា) ទម្រង់នៃជំងឺរលាកសុដន់ដែលសត្វគោមើលទៅមានសុខភាពល្អធម្មតា ហើយទឹកដោះមិនមានការប្រែប្រួលរូបរាងខាងក្រៅ ប៉ុន្តែមានផ្ទុកបាក់តេរីបង្កជំងឺដែលអាចធ្វើឱ្យទិន្នផលទឹកដោះធ្លាក់ចុះជាលំដាប់ដោយស្ងាត់ៗ។ វាដូចជាជំងឺលើសឈាមក្នុងមនុស្ស ដែលយើងមើលមិនឃើញពីរោគសញ្ញាខាងក្រៅ ប៉ុន្តែវាបំផ្លាញសុខភាពនិងកាត់បន្ថយកម្លាំងបន្តិចម្តងៗ។
Repeat breeding (ការបង្កាត់ពូជមិនជាប់ / ការបង្កាត់ពូជផ្ទួនៗ) បញ្ហាបន្តពូជដែលមេគោមិនអាចមានគភ៌ទោះបីជាបានទទួលការបង្កាត់ពូជ (សិប្បនិម្មិតឬធម្មជាតិ) ៣ ដង ឬច្រើនដងក៏ដោយ ដែលបណ្តាលឱ្យកសិករខាតបង់ពេលវេលា ប្រាក់កម្រៃបង្កាត់ និងបាត់បង់ទិន្នផលទឹកដោះព្រោះមេគោពន្យារពេលបង្កើតកូន។ ដូចជាការសាបព្រោះគ្រាប់ពូជនៅលើដីជាច្រើនដង ប៉ុន្តែគ្រាប់ពូជទាំងនោះនៅតែមិនព្រមដុះពន្លកសោះ។
Animal insurance (ការធានារ៉ាប់រងសត្វពាហនៈ) កម្មវិធីហិរញ្ញវត្ថុដែលរដ្ឋាភិបាល ឬក្រុមហ៊ុនឯកជនជួយចេញសំណងដល់កសិករនៅពេលដែលសត្វរបស់ពួកគេងាប់ដោយសារជំងឺឆ្លងធ្ងន់ធ្ងរ (ដូចជាជំងឺអុតក្តាម) ឬគ្រោះមហន្តរាយ ដើម្បីជួយសង្គ្រោះនិងកាត់បន្ថយការខាតបង់សេដ្ឋកិច្ចរបស់កសិដ្ឋាន។ ដូចជាការបង់ប្រាក់សន្សំទុកមុនបន្តិចបន្តួច ដើម្បីឱ្យគេជួយចេញលុយសងយើងវិញនៅពេលរថយន្តរបស់យើងជួបគ្រោះថ្នាក់ចៃដន្យ។
Coagulase-positive staphylococci / CPS (បាក់តេរីស្តាហ្វីឡូកូកមានអង់ស៊ីមកូអាហ្គូឡាសវិជ្ជមាន) ក្រុមបាក់តេរីបង្កជំងឺឆ្លង (ដូចជា Staphylococcus aureus) ដែលអាចបញ្ចេញអង់ស៊ីមម៉្យាងហៅថា Coagulase ធ្វើឱ្យឈាមកក ហើយវាគឺជាភ្នាក់ងារចម្បងដែលបង្កជំងឺរលាកសុដន់រ៉ាំរ៉ៃនៅក្នុងហ្វូងគោទឹកដោះ។ ដូចជាក្រុមចោរដែលមានអាវុធពិសេសដែលអាចធ្វើឱ្យប្រព័ន្ធការពារផ្ទះ (រាងកាយសត្វ) គាំងដំណើរការ ទើបពួកគេអាចលួចចូលទៅបំផ្លាញបានយ៉ាងងាយ។
Intra-class correlation coefficient / ICC (មេគុណសហសម្ព័ន្ធក្នុងថ្នាក់) រង្វាស់ស្ថិតិដែលប្រើដើម្បីវាយតម្លៃភាពស្រដៀងគ្នានៃទិន្នន័យនៅក្នុងក្រុមតែមួយ (ឧទាហរណ៍ គោទាំងអស់ដែលរស់នៅក្នុងកសិដ្ឋានតែមួយ) ដែលជួយដល់អ្នកស្រាវជ្រាវក្នុងការគណនាទំហំសំណាកឱ្យបានត្រឹមត្រូវ និងជៀសវាងភាពលម្អៀង។ វាជារង្វាស់មួយដើម្បីប្រាប់ថា តើសិស្សនៅក្នុងថ្នាក់តែមួយមានពិន្ទុប្រហាក់ប្រហែលគ្នាកម្រិតណា បើធៀបនឹងសិស្សនៅថ្នាក់ផ្សេងទៀត។
Endemic disease (ជំងឺរ៉ាំរ៉ៃប្រចាំតំបន់ / ជំងឺអង់ដេមិច) ជំងឺដែលកើតមានជាប្រចាំ និងរ៉ាំរ៉ៃនៅក្នុងតំបន់ភូមិសាស្ត្រ ឬក្នុងហ្វូងសត្វណាមួយ (ដូចជាជំងឺរលាកសុដន់នៅក្នុងតំបន់ចិញ្ចឹមគោ) ជាជាងការផ្ទុះឡើងជាទ្រង់ទ្រាយធំមួយរំពេចហើយបាត់ទៅវិញ។ ដូចជាជំងឺផ្តាសាយដែលតែងតែកើតមានជារឿយៗនៅរដូវភ្លៀងជារៀងរាល់ឆ្នាំ មិនមែនជាជំងឺរាតត្បាតថ្មីដែលទើបតែលេចឡើងនោះទេ។
Polymerase Chain Reaction / PCR (ប្រតិកម្មច្រវ៉ាក់ប៉ូលីមេរ៉ាស) បច្ចេកទេសមន្ទីរពិសោធន៍ដែលថតចម្លង និងពង្រីកបំណែកសេនេទិច (DNA) របស់មេរោគ ដើម្បីបញ្ជាក់អត្តសញ្ញាណបាក់តេរីបង្កជំងឺឱ្យបានច្បាស់លាស់បំផុត ទោះបីជាមានមេរោគក្នុងបរិមាណតិចតួចក៏ដោយ។ វាដូចជាម៉ាស៊ីនថតចម្លងនិងពង្រីករូបភាព (Zoom) ដែលអាចពង្រីករូបថតព្រាលៗដ៏តូចមួយឱ្យធំនិងច្បាស់ ដើម្បីងាយស្រួលមើលថាតើវាជារូបអ្នកណា។
Wilcoxon rank-sum test (តេស្តស្ថិតិ Wilcoxon rank-sum) វិធីសាស្ត្រស្ថិតិដែលប្រើសម្រាប់ប្រៀបធៀបមធ្យមភាគនៃក្រុមឯករាជ្យពីរ (ឧ. កសិដ្ឋានដែលមានទិញធានារ៉ាប់រង និងកសិដ្ឋានគ្មានទិញធានារ៉ាប់រង) នៅពេលដែលទិន្នន័យមិនមានរបាយធម្មតា (Non-normal distribution) ។ វាជាឧបករណ៍គណិតវិទ្យាសម្រាប់វាស់ស្ទង់ថា តើក្រុមមនុស្សពីរក្រុមពិតជាមានភាពខុសគ្នាឬអត់ ដោយមិនខ្វល់ថាទិន្នន័យរបស់ពួកគេតម្រៀបគ្នាយ៉ាងណានោះទេ។
