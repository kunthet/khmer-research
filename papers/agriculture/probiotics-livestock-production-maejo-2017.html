ឯកសារនេះគឺជាអត្ថបទពិនិត្យឡើងវិញ (Review Article) ពីសាកលវិទ្យាល័យម៉ែចូ (Maejo University) ប្រទេសថៃ ដែលបានចងក្រងការស្រាវជ្រាវពាក់ព័ន្ធនឹងបសុសត្វជាច្រើនកន្លែង។ ដោយសារប្រទេសថៃ និងកម្ពុជាមានអាកាសធាតុ លក្ខខណ្ឌកសិកម្ម និងប្រភេទពូជសត្វចិញ្ចឹមស្រដៀងគ្នា ទិន្នន័យទាំងនេះមានភាពពាក់ព័ន្ធ និងអាចយកមកអនុវត្តបានយ៉ាងមានប្រសិទ្ធភាពនៅកម្ពុជា។

លទ្ធភាពនៃការអនុវត្ត (Applicability)៖

វិធីសាស្ត្រនៃការប្រើប្រាស់ប្រូបាយអូទិកនេះ គឺមានសារៈសំខាន់ និងអាចយកមកអនុវត្តជាក់ស្តែងបានយ៉ាងល្អសម្រាប់វិស័យកសិកម្មនិងការចិញ្ចឹមសត្វនៅកម្ពុជា។

  • កសិដ្ឋានចិញ្ចឹមជ្រូកពាណិជ្ជកម្ម (ឧ. តំបន់ជាយក្រុងភ្នំពេញ តាកែវ និងកំពង់ស្ពឺ): អាចប្រើប្រាស់ដើម្បីកាត់បន្ថយអត្រាស្លាប់របស់កូនជ្រូកដោយសារជំងឺរាករូស (E. coli) ជួយឱ្យជ្រូកឆាប់ធាត់ កាត់បន្ថយការខាតបង់ចំណី និងកាត់បន្ថយក្លិនស្អុយនៃលាមកជ្រូកដែលរំខានដល់អ្នកភូមិ។
  • កសិដ្ឋានចិញ្ចឹមមាន់យកសាច់ និងពង (ឧ. ខេត្តកណ្តាល និងកំពង់ចាម): អាចប្រើដើម្បីបង្កើនទម្ងន់ស៊ុត កាត់បន្ថយកម្រិតកូឡេស្តេរ៉ុលក្នុងស៊ុត និងផលិតសាច់មាន់ប្រកបដោយសុវត្ថិភាព គ្មានសំណល់ថ្នាំអង់ទីប៊ីយ៉ូទិក ដែលឆ្លើយតបទៅនឹងតម្រូវការទីផ្សារក្នុងស្រុក។
  • ការចិញ្ចឹមគោទឹកដោះ និងគោសាច់កសិដ្ឋាន: ការប្រើប្រាស់ប្រូបាយអូទិកប្រភេទមេដំបែ (Yeast) ជួយសម្រួលដល់ការរំលាយអាហារប្រភេទសរសៃ (Cellulose) នៅក្នុងក្រពះគោ ជួយឱ្យគោស៊ីចំណីបានច្រើន ឆាប់ធាត់ និងបង្កើនទិន្នផលទឹកដោះ។

ការងាកមកប្រើប្រាស់ប្រូបាយអូទិកជំនួសថ្នាំអង់ទីប៊ីយ៉ូទិក មិនត្រឹមតែជួយលើកកម្ពស់សុខភាពសត្វ និងសេដ្ឋកិច្ចកសិករកម្ពុជាប៉ុណ្ណោះទេ ថែមទាំងធានាបាននូវសុវត្ថិភាពចំណីអាហារសាធារណៈផងដែរ។

៤. ផែនការសកម្មភាពសម្រាប់និស្សិត (Actionable Roadmap)

ដើម្បីអនុវត្តតាមការសិក្សានេះ និស្សិតគួរអនុវត្តតាមជំហានខាងក្រោម៖

  1. សិក្សាស្វែងយល់អំពីប្រភេទប្រូបាយអូទិក និងយន្តការរបស់វា: សិក្សាពីលក្ខណៈសម្បត្តិនៃបាក់តេរី និងមេដំបែដែលមានប្រយោជន៍ដូចជា Lactobacillus, Bacillus subtilis ឬ Saccharomyces cerevisiae តាមរយៈឯកសារស្រាវជ្រាវ និងទិន្នន័យពីក្រសួងកសិកម្មរុក្ខាប្រមាញ់ និងនេសាទ។
  2. ជ្រើសរើសប្រភេទប្រូបាយអូទិកឱ្យស្របតាមប្រភេទសត្វចិញ្ចឹម: វាយតម្លៃនិងជ្រើសរើសផលិតផលពាណិជ្ជកម្មដោយផ្អែកលើប្រភេទសត្វ (ឧ. ជ្រូក គោ មាន់) ដោយត្រូវប្រាកដថាអតិសុខុមប្រាណទាំងនោះមានលក្ខណៈសម្បត្តិធន់នឹងអាស៊ីតក្រពះសត្វ និងអំបិលប្រមាត់ (Bile salts) តាមស្តង់ដារគុណភាព។
  3. សាកល្បងប្រើប្រាស់ក្នុងកសិដ្ឋានខ្នាតតូចជាមុន (Pilot Testing): អនុវត្តការលាយប្រូបាយអូទិកទៅក្នុងចំណីសត្វ (លាយក្នុងចំណីស្ងួត ឬបាញ់ស្ព្រាយ៍) ក្នុងកម្រិតដែលបានណែនាំដោយក្រុមហ៊ុនផលិត ដោយត្រូវប្រុងប្រយ័ត្នដាច់ខាតមិនឱ្យប្រើប្រាស់រួមគ្នាជាមួយថ្នាំអង់ទីប៊ីយ៉ូទិកឡើយ។
  4. កត់ត្រាទិន្នន័យ និងប្រៀបធៀបប្រសិទ្ធភាព: ធ្វើការកត់ត្រា និងតាមដានអត្រាកំណើនទម្ងន់ (Average Daily Gain) អត្រាបំប្លែងចំណី (Feed Conversion Ratio) អត្រាស្លាប់ និងការថយចុះនៃក្លិនលាមកសត្វ ដើម្បីប្រៀបធៀបជាមួយនឹងកាលដែលមិនទាន់ប្រើប្រាស់។
  5. ពង្រីកការប្រើប្រាស់ និងចងក្រងជាស្តង់ដារកសិកម្មសរីរាង្គ (SOP): បន្ទាប់ពីទទួលបានលទ្ធផលវិជ្ជមាន ចងក្រងរបាយការណ៍ជាគោលការណ៍ណែនាំស្តង់ដារប្រតិបត្តិ (SOP) សម្រាប់ការចិញ្ចឹមសត្វសរីរាង្គ និងចែករំលែកបទពិសោធន៍បច្ចេកទេសនេះទៅកាន់សហគមន៍កសិករដទៃទៀត។

៥. វាក្យសព្ទបច្ចេកទេស (Technical Glossary)

ពាក្យបច្ចេកទេស ការពន្យល់ជាខេមរភាសា (Khmer Explanation) និយមន័យសាមញ្ញ (Simple Definition)
Probiotics អតិសុខុមប្រាណ (បាក់តេរី ឬមេដំបែ) មានប្រយោជន៍ ដែលនៅពេលផ្តល់ឱ្យសត្វស៊ីក្នុងបរិមាណសមស្រប វាជួយកែលម្អតុល្យភាពមីក្រូជីវសាស្ត្រនៅក្នុងពោះវៀន ជួយដល់ការរំលាយអាហារ និងជំរុញប្រព័ន្ធភាពស៊ាំ។ ដូចជាទាហានល្អប្រចាំការក្នុងពោះវៀន ដែលជួយការពាររាងកាយពីមេរោគអាក្រក់ និងជួយកិនរំលាយអាហារ។
Metabolite សមាសធាតុគីមីដែលផលិតឡើងដោយអតិសុខុមប្រាណក្នុងកំឡុងពេលដំណើរការមេតាបូលីស (ការរំលាយអាហារ) ដូចជា អាស៊ីតឡាក់ទិក វីតាមីន ឬអង់ស៊ីម ដែលផ្តល់ប្រយោជន៍ដោយផ្ទាល់ដល់រាងកាយសត្វ។ ដូចជារោងចក្រតូចៗនៅក្នុងពោះវៀន ដែលផលិតវីតាមីននិងថ្នាំបំប៉នចេញពីចំណីដែលសត្វស៊ី។
Bacteriocin ប្រូតេអ៊ីន ឬផិបទីតម្យ៉ាង ដែលផលិតដោយបាក់តេរីល្អ (ប្រូបាយអូទិក) ដែលមានសមត្ថភាពអាចរារាំង ការពារ ឬសម្លាប់បាក់តេរីបង្កជំងឺផ្សេងៗនៅក្នុងពោះវៀនសត្វ។ ដូចជាអាវុធជីវសាស្ត្រ ឬគ្រាប់កាំភ្លើងដែលទាហានល្អប្រើដើម្បីបាញ់កម្ចាត់សត្រូវ (មេរោគអាក្រក់)។
Average daily gain (ADG) រង្វាស់នៃការកើនឡើងទម្ងន់ជាមធ្យមរបស់សត្វក្នុងមួយថ្ងៃ ដែលជាសូចនាករសំខាន់សម្រាប់វាយតម្លៃពីភាពជោគជ័យនៃការលូតលាស់នៅក្នុងកសិកម្មចិញ្ចឹមសត្វយកសាច់។ ដូចជាការថ្លឹងទម្ងន់ក្មេងរៀងរាល់ថ្ងៃ ដើម្បីមើលថាគេធំធាត់លឿនប៉ុណ្ណាក្នុងមួយថ្ងៃៗ។
Feed conversion ratio (FCR) សមាមាត្ររវាងបរិមាណចំណីសរុបដែលសត្វស៊ី និងបរិមាណទម្ងន់ដែលសត្វកើនឡើង (ឧទាហរណ៍៖ ស៊ីចំណីប៉ុន្មានគីឡូ ទើបឡើងសាច់បាន១គីឡូ)។ អត្រា FCR កាន់តែទាប មានន័យថាសត្វរំលាយអាហារនិងលូតលាស់កាន់តែល្អ ដែលជួយសន្សំសំចៃចំណាយ។ ដូចជាការវាស់ស្ទង់ថាតើឡានមួយស៊ីសាំងអស់ប៉ុន្មានលីត្រ ដើម្បីអាចរត់បានចម្ងាយ១០០គីឡូម៉ែត្រអញ្ចឹងដែរ។
Immunoglobulin A (IgA) អង់ទីករ (Antibody) ដ៏សំខាន់មួយនៅក្នុងប្រព័ន្ធភាពស៊ាំ ដែលមានវត្តមានច្រើនជាងគេនៅក្នុងទឹករំអិលពោះវៀន និងទឹកដោះទឹកដំបូង (Colostrum) ដើម្បីការពារការជ្រៀតចូលនៃមេរោគតាមជញ្ជាំងកោសិកា។ ដូចជាខែលការពារ ឬអ្នកយាមទ្វារនៅតាមជញ្ជាំងពោះវៀននិងក្នុងទឹកដោះម្តាយ ដែលរារាំងមិនឱ្យមេរោគឆ្លងចូលទៅក្នុងឈាមបាន។
Villi រោមតូចៗរាប់លាននៅលើជញ្ជាំងពោះវៀនតូច ដែលមានតួនាទីពង្រីកផ្ទៃក្រឡាដើម្បីស្រូបយកសារធាតុចិញ្ចឹមពីចំណីអាហារទៅចិញ្ចឹមរាងកាយ។ ការប្រើប្រូបាយអូទិកជួយធ្វើឱ្យរោមទាំងនេះដុះវែងនិងមានសុខភាពល្អ។ ដូចជាសរសៃរោមតូចៗនៅលើកន្សែងពោះគោ ដែលជួយស្រូបទឹកបានច្រើននិងលឿនជាងក្រណាត់រលោងធម្មតា។
