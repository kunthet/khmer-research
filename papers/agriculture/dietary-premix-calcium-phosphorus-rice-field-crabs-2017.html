  • ខេត្តនៅជាប់តំបន់ព្រំដែនថៃ និងភាគពាយ័ព្យ (ឧ. បន្ទាយមានជ័យ បាត់ដំបង): ការចិញ្ចឹមក្តាមប្រភេទ Esanthelphusa dugasti អាចទទួលបានជោគជ័យខ្ពស់នៅតំបន់ទាំងនេះ ដោយសារបរិស្ថានធម្មជាតិស្រដៀងគ្នានឹងទីតាំងសិក្សា។
  • វិស័យវារីវប្បកម្មខ្នាតតូច (Small-scale Aquaculture): កសិករអាចផលិតចំណីដោយខ្លួនឯង ដោយប្រើប្រាស់វត្ថុធាតុដើមក្នុងស្រុក (កន្ទក់ ចុងអង្ករ ម្សៅត្រី) រួមផ្សំជាមួយការទិញសារធាតុវីតាមីនបន្ថែម (Premix) ដើម្បីទទួលបានទិន្នផលខ្ពស់ និងកាត់បន្ថយការពឹងផ្អែកលើការចាប់ក្តាមពីធម្មជាតិ។
  • សុវត្ថិភាពចំណីអាហារ និងសុខភាពសាធារណៈ (Food Safety): ការចិញ្ចឹមក្តាមក្នុងកសិដ្ឋាន (ទឹកស្អាត និងចំណីត្រឹមត្រូវ) ជួយកាត់បន្ថយហានិភ័យនៃការឆ្លងប៉ារ៉ាស៊ីតតេនញ៉ាសួត (Paragonimus spp.) និងសារធាតុគីមីពីថ្នាំសម្លាប់សត្វល្អិត ដែលជារឿយៗជួបប្រទះលើក្តាមចាប់ពីស្រែ។

ការអនុវត្តរូបមន្តចំណីនេះអាចជួយបង្កើនទិន្នផលក្តាមស្រែពាណិជ្ជកម្មប្រកបដោយនិរន្តរភាពនៅកម្ពុជា ឆ្លើយតបទៅនឹងតម្រូវការទីផ្សារ និងធានាសុវត្ថិភាពសម្រាប់អ្នកបរិភោគ។

៤. ផែនការសកម្មភាពសម្រាប់និស្សិត (Actionable Roadmap)

ដើម្បីអនុវត្តតាមការសិក្សានេះ និស្សិតគួរអនុវត្តតាមជំហានខាងក្រោម៖

  1. សិក្សារូបមន្ត និងរៀបចំវត្ថុធាតុដើមចំណី: អ្នកស្រាវជ្រាវ ឬសិស្សត្រូវប្រើប្រាស់ Microsoft Excel ដើម្បីគណនាសមាមាត្រប្រូតេអ៊ីន (ប្រហែល ៣០-៣៣%) ពីវត្ថុធាតុដើមក្នុងស្រុក និងត្រូវគណនាបរិមាណកាល់ស្យូម-ផូស្វ័រ ឱ្យបានច្បាស់លាស់ក្នុងសមាមាត្រ ២.៤:១ ដោយពឹងផ្អែកលើទិន្នន័យ (Proximate analysis)។
  2. សាកល្បងផលិតចំណីកែច្នៃខ្នាតតូច: ទិញសារធាតុ Vitamin/Mineral Premix ពីទីផ្សារបសុពេទ្យ រួចលាយជាមួយម្សៅវត្ថុធាតុដើមក្នុងអត្រា ៣ ក្រាម/គីឡូក្រាម។ បន្ទាប់មក ប្រើប្រាស់ម៉ាស៊ីនកិនសាច់ ឬម៉ាស៊ីនសង្កត់គ្រាប់ចំណី (Pelletizer) ដើម្បីផលិតគ្រាប់ចំណីទំហំ ២មម រួចយកទៅសម្ងួតក្នុងឡ (Electric Oven) នៅសីតុណ្ហភាព ៦០ អង្សាសេ។
  3. រៀបចំប្រព័ន្ធចិញ្ចឹម និងការគ្រប់គ្រងទឹក: រៀបចំអាងចិញ្ចឹម ដោយប្រើធុងប្លាស្ទិក ឬ Acrylic tanks ដែលមានបំពាក់ប្រព័ន្ធបង្ហូរទឹកចេញចូល។ ត្រូវតាមដាន និងរក្សាគុណភាពទឹកដោយប្រើប្រាស់ Water Quality Test Kit (រក្សា pH ៧.១-៧.៩, និងអុកស៊ីសែនរលាយ DO លើសពី ៣.៥ mg/L) ជាប្រចាំ។
  4. ប្រមូលទិន្នន័យលូតលាស់ និងវិភាគលទ្ធផល: ប្រើប្រាស់បន្ទាត់កាលីប (Vernier Caliper) និងជញ្ជីងឌីជីថល (Digital Scale) ដើម្បីវាស់ប្រវែងសំបក និងទម្ងន់ក្តាមរៀងរាល់សប្តាហ៍។ យកទិន្នន័យដែលបានកត់ត្រាទៅវិភាគរក SGR និង FCR តាមរយៈកម្មវិធី SPSS ឬ R Studio ដើម្បីទាញសេចក្តីសន្និដ្ឋាន។

៥. វាក្យសព្ទបច្ចេកទេស (Technical Glossary)

ពាក្យបច្ចេកទេស ការពន្យល់ជាខេមរភាសា (Khmer Explanation) និយមន័យសាមញ្ញ (Simple Definition)
Dietary premix (សារធាតុចម្រុះបន្ថែមក្នុងរបបអាហារ) គឺជាល្បាយនៃវីតាមីន និងសារធាតុរ៉ែដែលត្រូវបានគេរៀបចំទុកជាមុន ដើម្បីយកទៅលាយបញ្ចូលក្នុងចំណីសត្វក្នុងបរិមាណតិចតួច ដែលជួយធានាថាសត្វទទួលបានមីក្រូសារធាតុចិញ្ចឹមគ្រប់គ្រាន់សម្រាប់ការលូតលាស់ និងសុខភាពរឹងមាំ។ ដូចជាការលេបថ្នាំវីតាមីនចម្រុះប្រចាំថ្ងៃ ដើម្បីបំពេញបន្ថែមនូវជីវជាតិដែលយើងមិនទទួលបានគ្រប់គ្រាន់ពីការញ៉ាំអាហារធម្មតា។
Carapace (ស្នូកក្តាម) គឺជាសំបកផ្នែកខាងលើដ៏រឹងរបស់សត្វអត់ឆ្អឹងកង (ដូចជាក្តាម ឬបង្គា) ដែលមានតួនាទីយ៉ាងសំខាន់ក្នុងការការពារសរីរាង្គទន់ៗខាងក្នុង និងផ្តល់នូវទម្រង់រាងកាយដល់សត្វទាំងនោះ។ ដូចជាមួកសុវត្ថិភាព ឬអាវក្រោះដ៏រឹងមាំដែលសត្វពាក់ដើម្បីការពាររាងកាយពីការប៉ះទង្គិច និងការវាយប្រហារផ្សេងៗ។
Molting frequency (ប្រេកង់នៃការសកសំបក) គឺជាចំនួនដង ឬអត្រាដែលសត្វអត់ឆ្អឹងកង (ដូចជាក្តាម) ធ្វើការជំរុះសំបកចាស់ចោល ដើម្បីបង្កើតសំបកថ្មីដែលមានទំហំធំជាងមុន ក្នុងរយៈពេលកំណត់ណាមួយ។ នេះគឺជាដំណើរការចាំបាច់បំផុតដើម្បីឱ្យក្តាមអាចលូតលាស់ទំហំខ្លួនបាន។ ដូចជាក្មេងដែលកំពុងធំធាត់ ត្រូវតែផ្លាស់ប្តូរខោអាវទំហំធំជាងមុនជារឿយៗ នៅពេលដែលខោអាវចាស់តូចចង្អៀតលែងស្លៀកកើត។
Specific growth rate (SGR) (អត្រាលូតលាស់ជាក់លាក់) គឺជារង្វាស់គណិតវិទ្យាក្នុងវារីវប្បកម្មដែលប្រើសម្រាប់គណនាពីភាគរយនៃកំណើនទម្ងន់ខ្លួនរបស់សត្វក្នុងមួយថ្ងៃ ធៀបនឹងរយៈពេលនៃការផ្តល់ចំណីសរុប ដើម្បីវាយតម្លៃពីភាពជោគជ័យនៃការចិញ្ចឹម។ ដូចជាការគណនាអត្រាការប្រាក់ប្រចាំថ្ងៃដែលធនាគារផ្តល់ឱ្យ ប៉ុន្តែនេះគឺជាការគណនាភាគរយនៃទម្ងន់សត្វដែលកើនឡើងជារៀងរាល់ថ្ងៃ។
Feed conversion ratio (FCR) (អត្រាបំប្លែងចំណី) គឺជាសូចនាករវាស់ស្ទង់ពីប្រសិទ្ធភាពរបស់សត្វក្នុងការបំប្លែងចំណីដែលវាស៊ី ទៅជាទម្ងន់ខ្លួនដែលកើនឡើង។ តម្លៃ FCR កាន់តែទាប មានន័យថាសត្វនោះស៊ីចំណីតិច ប៉ុន្តែឆាប់ធំ ដែលផ្តល់ផលចំណេញច្រើនដល់កសិករ។ ដូចជាកម្រិតស៊ីសាំងរបស់ម៉ូតូ (១លីត្រជិះបានប៉ុន្មានគីឡូម៉ែត្រ) បើម៉ូតូស៊ីសាំងតិចតែជិះបានឆ្ងាយ គឺមានប្រសិទ្ធភាពខ្ពស់ ដូចសត្វស៊ីចំណីតិចតែឆាប់ធំ។
Calcium and Phosphorus ratio (សមាមាត្រកាល់ស្យូមនិងផូស្វ័រ) គឺជាការប្រៀបធៀបបរិមាណរវាងសារធាតុកាល់ស្យូម និងផូស្វ័រនៅក្នុងរបបអាហារ។ ការរក្សាសមាមាត្រនេះឱ្យបានត្រឹមត្រូវ (ឧ. ២.៤:១) គឺមានសារៈសំខាន់ណាស់ ព្រោះរ៉ែទាំងពីរនេះធ្វើការរួមគ្នាដើម្បីបង្កើតសំបកក្តាមឱ្យបានរឹងមាំក្រោយពេលសកសំបករួច។ ដូចជាការលាយស៊ីម៉ងត៍និងខ្សាច់ក្នុងអត្រាត្រឹមត្រូវ ដើម្បីចាក់សសរផ្ទះឱ្យរឹងមាំ បើលាយខុសមាត្រដ្ឋាន សសរនឹងស្រួយងាយបាក់។
Exoskeleton (គ្រោងឆ្អឹងក្រៅ / សំបកក្រៅ) គឺជាគម្របរឹងខាងក្រៅរាងកាយរបស់សត្វមួយចំនួន (ដូចជាសត្វល្អិត និងក្តាម) ដែលផ្សំឡើងពីសារធាតុ Chitin និងរ៉ែដូចជាកាល់ស្យូម ដើម្បីទ្រទ្រង់ទម្ងន់ខ្លួន និងការពារសរីរាង្គខាងក្នុង។ ដូចជាសំបកកង់ឡានដ៏រឹងមាំ ដែលជួយទ្រទ្រង់ខ្យល់ខាងក្នុង និងការពារកង់ពីការមុតធ្លាយ។
Esanthelphusa dugasti (ពូជក្តាមស្រែប្រភេទ Esanthelphusa dugasti) គឺជាឈ្មោះវិទ្យាសាស្ត្ររបស់ពូជក្តាមស្រែទឹកសាបមួយប្រភេទ ដែលមានប្រភព និងត្រូវបានគេរកឃើញយ៉ាងទូលំទូលាយនៅតំបន់អាស៊ីអាគ្នេយ៍ ជាពិសេសនៅភាគឦសាននៃប្រទេសថៃ និងមានតម្លៃសេដ្ឋកិច្ចខ្ពស់សម្រាប់ការបរិភោគជាប្រចាំថ្ងៃ។ ដូចជាពូជស្រូវផ្ការំដួល ដែលជាពូជពិសេសប្រចាំតំបន់ និងមានការពេញនិយមខ្ពស់នៅលើទីផ្សារ។
