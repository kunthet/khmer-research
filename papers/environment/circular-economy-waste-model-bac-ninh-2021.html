ស្ថាប័នស្រាវជ្រាវ និងនវានុវត្តន៍ (Research & Innovation Institutions) ស្រាវជ្រាវ និងផ្ទេរបច្ចេកវិទ្យាទំនើបសម្រាប់ការកែច្នៃកាកសំណល់ (Waste-to-energy, modern recycling tech) ដើម្បីជំនួសបច្ចេកវិទ្យាចាស់ៗនៅតាមភូមិសិប្បកម្ម។ មធ្យម (Medium)

៤. បរិបទកម្ពុជា (Cambodia Context)

របាយការណ៍នេះមានសារៈសំខាន់ខ្លាំងសម្រាប់ប្រទេសកម្ពុជា ខណៈដែលកម្ពុជាកំពុងពង្រីកតំបន់សេដ្ឋកិច្ចពិសេស (SEZ) របស់ខ្លួនជាច្រើន។ ការរៀនសូត្រពីគំរូសេដ្ឋកិច្ចវិលជុំរបស់ប្រទេសវៀតណាមអាចជួយកម្ពុជាជៀសវាងបញ្ហាបំពុលបរិស្ថាន និងបង្វែរកាកសំណល់ទៅជាធនធានសេដ្ឋកិច្ចវិញ។

ផលប៉ះពាល់មូលដ្ឋាន (Local Implications)៖

  • ឧស្សាហកម្ម និងតំបន់សេដ្ឋកិច្ចពិសេស (Industry & SEZ): ក្រុមប្រឹក្សាអភិវឌ្ឍន៍កម្ពុជា (CDC) និងក្រសួងពាក់ព័ន្ធអាចជំរុញការអនុវត្តគំរូ តំបន់ឧស្សាហកម្មអេកូឡូស៊ី (Eco-industrial park) នៅតំបន់សេដ្ឋកិច្ចពិសេសក្រុងព្រះសីហនុ ភ្នំពេញ ធានាឱ្យមានបណ្តាញកែច្នៃកាកសំណល់រួមគ្នា។
  • ការគ្រប់គ្រងបរិស្ថាន (Environmental Management): ក្រសួងបរិស្ថាន (MoE) អាចរៀបចំគោលនយោបាយទំនួលខុសត្រូវបន្ថែមរបស់អ្នកផលិត (EPR - Extended Producer Responsibility) ដើម្បីដាក់បន្ទុកលើរោងចក្រក្នុងការកែច្នៃសំណល់របស់ខ្លួនឡើងវិញ ជំនួសឱ្យការចាក់ចោលនៅទីលាន។
  • សហគ្រាសធុនតូច និងមធ្យម (SME Sector): ក្រសួងឧស្សាហកម្ម វិទ្យាសាស្ត្រ បច្ចេកវិទ្យា និងនវានុវត្តន៍ (MISTI) អាចគាំទ្រការធ្វើឱ្យប្រសើរឡើងនូវបច្ចេកវិទ្យាកែច្នៃសម្រាប់សិប្បកម្មក្នុងស្រុក (អេតចាយកែច្នៃ) ដើម្បីកាត់បន្ថយផលប៉ះពាល់បរិស្ថាន។

ការអនុវត្តគំរូសេដ្ឋកិច្ចវិលជុំនេះ អាចជួយកម្ពុជាឱ្យប្រែក្លាយបញ្ហាកាកសំណល់ឧស្សាហកម្មទៅជាវត្ថុធាតុដើមដ៏មានតម្លៃ កាត់បន្ថយការបំពុល និងធានាបាននូវការអភិវឌ្ឍឧស្សាហកម្មប្រកបដោយចីរភាព និងពណ៌បៃតង។

៥. ផែនការអនុវត្ត (Implementation Roadmap)

ដើម្បីអនុវត្តតាមអនុសាសន៍នៃរបាយការណ៍នេះ គួរអនុវត្តតាមជំហានខាងក្រោម៖

  1. វាយតម្លៃ និងធ្វើសវនកម្មកាកសំណល់ (Waste Assessment and Auditing): កំណត់ប្រភេទ និងបរិមាណកាកសំណល់រឹងឧស្សាហកម្មនៅតាមតំបន់សេដ្ឋកិច្ចពិសេស (SEZ) នីមួយៗ តាមរយៈការធ្វើសវនកម្មកាកសំណល់ (Waste Audit) ដើម្បីស្វែងរកសក្តានុពលនៃការកែច្នៃលោហៈ ប្លាស្ទិក និងក្រដាស។
  2. បង្កើតក្របខ័ណ្ឌគតិយុត្ត (Develop Legal Framework): ក្រសួងបរិស្ថាន និងស្ថាប័នពាក់ព័ន្ធត្រូវរៀបចំលិខិតបទដ្ឋានគតិយុត្ត និងគោលនយោបាយលើកទឹកចិត្ត (EPR, Tax Incentives) សម្រាប់រោងចក្រដែលអនុវត្តផលិតកម្មស្អាត និងការកែច្នៃ។
  3. អនុវត្តគម្រោងសាកល្បងកម្រិតរោងចក្រ (Implement Factory-Level Pilots): ជ្រើសរើសរោងចក្រគោលដៅមួយចំនួននៅក្នុងតំបន់សេដ្ឋកិច្ចពិសេស ដើម្បីសាកល្បងអនុវត្តស្តង់ដារ 5S, 3R និងផលិតកម្មស្អាត (Cleaner Production) ដោយមានការគាំទ្របច្ចេកទេស។
  4. ការរៀបចំបណ្តាញសហឧស្សាហកម្ម (Establish Industrial Symbiosis Networks): បង្កើតវេទិកា ឬយន្តការ ដើម្បីឱ្យរោងចក្រនៅក្នុងតំបន់សេដ្ឋកិច្ចពិសេសអាចផ្លាស់ប្តូរ ឬលក់កាកសំណល់របស់ខ្លួនធ្វើជាវត្ថុធាតុដើម (Inputs) សម្រាប់ផលិតកម្មនៃរោងចក្រផ្សេងទៀត។
  5. ការវិនិយោគលើបច្ចេកវិទ្យាកែច្នៃ និងកែច្នៃថាមពល (Invest in Recycling & Waste-to-Energy): ទាក់ទាញការវិនិយោគពីវិស័យឯកជន (តាមរយៈយន្តការ PPP) ក្នុងការសាងសង់រោងចក្រកែច្នៃកាកសំណល់ទំនើប និងរោងចក្របំប្លែងកាកសំណល់ទៅជាថាមពល (Waste-to-Energy) ដើម្បីដោះស្រាយសំណល់ដែលមិនអាចកែច្នៃបាន។

៦. វាក្យសព្ទបច្ចេកទេស (Technical Glossary)

ពាក្យបច្ចេកទេស (English) ការពន្យល់ជាខេមរភាសា (Khmer Explanation) និយមន័យសាមញ្ញ (Simple Definition)
Circular Economy ទស្សនាវទននៃការរៀបចំប្រព័ន្ធសេដ្ឋកិច្ច ដែលកាកសំណល់ពីចង្វាក់ផលិតកម្មមួយ ត្រូវបានប្រើប្រាស់ជាវត្ថុធាតុដើមសម្រាប់ចង្វាក់ផលិតកម្មមួយទៀត ដើម្បីលុបបំបាត់ការបោះចោល និងទាញយកតម្លៃអតិបរមាពីធនធានរហូតដល់អស់លទ្ធភាព។ ដូចជាការយកដបប្លាស្ទិកចាស់ៗទៅកែច្នៃធ្វើជាសរសៃអំបោះសម្រាប់ត្បាញសម្លៀកបំពាក់ ជំនួសឱ្យការបោះចោលក្នុងធុងសំរាម។
Extended Producer Responsibility (EPR) គោលនយោបាយបរិស្ថានដែលតម្រូវឱ្យក្រុមហ៊ុនផលិតទទួលខុសត្រូវលើវដ្តជីវិតទាំងមូលនៃផលិតផលរបស់ខ្លួន រួមទាំងការប្រមូល និងការកែច្នៃឡើងវិញនៅពេលផលិតផលនោះខូច ឬឈប់ប្រើប្រាស់ពីអតិថិជន។ ដូចជាក្រុមហ៊ុនលក់ទូរស័ព្ទ ត្រូវមានកាតព្វកិច្ចទទួលទិញ ឬប្រមូលទូរស័ព្ទចាស់ៗពីអតិថិជនយកទៅកែច្នៃឡើងវិញ ដើម្បីកុំឱ្យពុលបរិស្ថាន។
Industrial Symbiosis ការតភ្ជាប់រវាងរោងចក្រ ឬសហគ្រាសផ្សេងៗគ្នា ដោយប្រើប្រាស់កាកសំណល់ ឬអនុផលនៃរោងចក្រមួយ ជាវត្ថុធាតុដើម ឬថាមពលសម្រាប់រោងចក្រមួយទៀតនៅក្នុងតំបន់តែមួយ។ ដូចជារោងចក្រកាត់ដេរយកក្រណាត់សល់ ទៅលក់ឱ្យរោងចក្រផលិតស្បែកជើងធ្វើជាទ្រនាប់ខាងក្នុង ធ្វើឱ្យចំណេញទាំងសងខាង។
Eco-industrial Park តំបន់ឧស្សាហកម្មដែលត្រូវបានរៀបចំឡើងដោយបញ្ចូលគោលការណ៍អភិវឌ្ឍន៍ប្រកបដោយចីរភាព ដោយរោងចក្រទាំងអស់សហការគ្នាគ្រប់គ្រងធនធាន ថាមពល និងកាកសំណល់រួមគ្នា ដើម្បីកាត់បន្ថយផលប៉ះពាល់បរិស្ថាន។ ដូចជាភូមិមួយដែលអ្នកភូមិទាំងអស់ប្រើប្រាស់ប្រព័ន្ធចម្រោះទឹក និងប្រព័ន្ធថាមពលព្រះអាទិត្យរួមគ្នា ដើម្បីសន្សំសំចៃ និងការពារបរិស្ថាន។
Linear Economy ទម្រង់សេដ្ឋកិច្ចបែបប្រពៃណី ដែលពឹងផ្អែកលើការទាញយកធនធាន ការផលិត ការប្រើប្រាស់ និងការបោះចោលចុងក្រោយ ដោយមិនខ្វល់ពីការកែច្នៃឡើងវិញ ដែលធ្វើឱ្យធនធានឆាប់រេចរឹល។ ដូចជាការទិញថង់ប្លាស្ទិកយកមកដាក់អីវ៉ាន់ម្តង រួចក៏បោះចោលទៅក្នុងគំនរសំរាមរហូត។
Waste Audit ការវាស់វែង និងវិភាគលម្អិតអំពីបរិមាណ និងប្រភេទកាកសំណល់ដែលរោងចក្រមួយបង្កើតឡើង ដើម្បីស្វែងរកឱកាសក្នុងការកាត់បន្ថយ ការកែច្នៃ និងការសន្សំសំចៃថ្លៃដើម។ ដូចជាការកត់ត្រាបញ្ជីចំណាយប្រចាំខែក្នុងគ្រួសារ ដើម្បីដឹងថាយើងខ្ជះខ្ជាយលុយលើអ្វីខ្លះ ហើយរកវិធីកាត់បន្ថយការចំណាយនោះ។
3R (Reduce, Reuse, Recycle) គោលការណ៍គ្រឹះក្នុងការគ្រប់គ្រងកាកសំណល់ ដែលផ្តោតលើការកាត់បន្ថយការប្រើប្រាស់ធនធាន ការប្រើប្រាស់ឡើងវិញនូវរបស់របរដែលនៅល្អ និងការកែច្នៃកាកសំណល់ទៅជាផលិតផលថ្មី។ ដូចជាការយកដបទឹកបរិសុទ្ធដែលផឹករួច ទៅលាងទឹក រួចច្រកទឹកផឹកម្តងទៀត (Reuse) ជាជាងទិញដបថ្មីជារៀងរាល់ថ្ងៃ។
Material Flow Analysis វិធីសាស្ត្រវិភាគក្នុងការតាមដាន និងវាយតម្លៃលំហូរនៃវត្ថុធាតុដើម និងថាមពល ចូលនិងចេញពីប្រព័ន្ធសេដ្ឋកិច្ច ឬរោងចក្រ ដើម្បីកំណត់ទីតាំងដែលខ្ជះខ្ជាយ និងសក្តានុពលកែច្នៃឡើងវិញ។ ដូចជាការគូសផែនទីបង្ហាញពីប្រភពទឹកហូរចូលក្នុងស្រះ និងផ្លូវទឹកហូរចេញ ដើម្បីដឹងថាទឹកបាត់បង់ទៅណាខ្លះ។
